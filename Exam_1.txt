# Print numbers divisible by 2 and 5 (1–50)
for i in range(1, 51):

        print(i)

# Print reverse numbers from 20 to 1

    print(i)

# Print cube of numbers (1–10)

# Separate numbers >10 and <=10

a = [5, 15, 8, 20, 3, 11]

b = []
c = []

for i in a:
    if i > 10:
        b.append(i)
    else:
        c.append(i)

print("Greater than 10:", b)
print("Less or equal 10:", c)
